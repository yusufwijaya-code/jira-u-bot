# Nudge Google Chat App — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Nudge sends every Google Chat message as its own Chat App "Nudge". Each user picks the sender in Pengaturan: Nudge or the operator account (Yusuf Wijaya). If sending through the app fails, the message goes out through the operator account instead. Commands, slash commands and card buttons work inside the Nudge DM.

**Architecture:** New `chat_app` modules: DTOs, a repository over the new `ms_user` columns, a service-account REST client, a JWT verifier, and the event service/handler. New `chat_app_activation` module (status, sender choice, one-click activation). New `chat_delivery` router that implements the existing notifier interfaces and is wired in place of webhook/DM routing. Callers keep their signatures: the router is passed into both notifier slots, and `selfEmail` is set to `""`. Angular adds an activation banner in the dashboard shell, a "Pengirim pesan" card, and admin columns and stats.

**Tech Stack:** Go 1.26, gin, MySQL 8 (`database/sql`), `golang.org/x/oauth2` + `golang.org/x/oauth2/google` (already in `go.mod`), stdlib crypto for JWT. Angular (standalone components, signals), Jasmine/Karma.

**Spec:** `docs/tech-spec/chat-app/Tech Spec - Nudge Google Chat App.md`

```go
// Package chat_app_dto holds the Google Chat App shapes Nudge uses: the
// per-user activation state (ms_user columns), the HTTP event Google posts
// to /api/chat/events and the synchronous reply (Tech Spec "Nudge sebagai
// Google Chat App").
package chat_app_dto

import (
	"encoding/json"
	"strings"
	"time"
)

// Message senders a user can choose in Pengaturan.
const (
	SenderNudge    = "nudge"
	SenderOperator = "operator"
)

// ValidSender reports whether s is one of the two sender values.
func ValidSender(s string) bool { return s == SenderNudge || s == SenderOperator }

// State is a user's Chat App activation and sender choice. Sender nil =
// automatic (nudge when activated, operator otherwise).
type State struct {
	Email         string
	SpaceID       string
	ChatUserID    string
	ActivatedDate *time.Time
	Sender        *string
}

// Activated reports whether the user has a DM with the Nudge app.
func (s State) Activated() bool { return s.SpaceID != "" }

// Event types Nudge handles.
const (
	EventAddedToSpace     = "ADDED_TO_SPACE"
	EventRemovedFromSpace = "REMOVED_FROM_SPACE"
	EventMessage          = "MESSAGE"
	EventCardClicked      = "CARD_CLICKED"
)

// Event is the interaction event Google Chat posts to the HTTP endpoint.
type Event struct {
	Type    string   `json:"type"`
	Space   Space    `json:"space"`
	User    ChatUser `json:"user"`
	Message *Message `json:"message,omitempty"`
	Action  *Action  `json:"action,omitempty"`
}

type Space struct {
	Name      string `json:"name"`
	Type      string `json:"type,omitempty"`
	SpaceType string `json:"spaceType,omitempty"`
}

// IsDM covers both the legacy "type" and the newer "spaceType" field.
func (s Space) IsDM() bool { return s.Type == "DM" || s.SpaceType == "DIRECT_MESSAGE" }

type ChatUser struct {
	Name        string `json:"name"`
	Email       string `json:"email,omitempty"`
	DisplayName string `json:"displayName,omitempty"`
	Type        string `json:"type,omitempty"`
}

type Message struct {
	Name         string        `json:"name,omitempty"`
	Text         string        `json:"text,omitempty"`
	ArgumentText string        `json:"argumentText,omitempty"`
	SlashCommand *SlashCommand `json:"slashCommand,omitempty"`
	CardsV2      []CardWithID  `json:"cardsV2,omitempty"`
}

type SlashCommand struct {
	CommandID FlexString `json:"commandId"`
}

// FlexString accepts a JSON string or number (Google sends int64 ids as
// strings, but tolerating a bare number costs nothing).
type FlexString string

func (f *FlexString) UnmarshalJSON(b []byte) error {
	if len(b) > 0 && b[0] == '"' {
		var s string
		if err := json.Unmarshal(b, &s); err != nil {
			return err
		}
		*f = FlexString(s)
		return nil
	}
	*f = FlexString(strings.TrimSpace(string(b)))
	return nil
}

type Action struct {
	ActionMethodName string            `json:"actionMethodName"`
	Parameters       []ActionParameter `json:"parameters,omitempty"`
}

// Param returns the value of the action parameter key ("" when absent).
func (a *Action) Param(key string) string {
	if a == nil {
		return ""
	}
	for _, p := range a.Parameters {
		if p.Key == key {
			return p.Value
		}
	}
	return ""
}

type ActionParameter struct {
	Key   string `json:"key"`
	Value string `json:"value"`
}

// Response is the synchronous reply to an event; the zero value marshals
// to {} (no reply).
type Response struct {
	Text                 string          `json:"text,omitempty"`
	CardsV2              []CardWithID    `json:"cardsV2,omitempty"`
	ActionResponse       *ActionResponse `json:"actionResponse,omitempty"`
	PrivateMessageViewer *ChatUser       `json:"privateMessageViewer,omitempty"`
}

type ActionResponse struct {
	Type string `json:"type"`
}

const (
	ResponseNewMessage    = "NEW_MESSAGE"
	ResponseUpdateMessage = "UPDATE_MESSAGE"
)

// OutgoingMessage is a message Nudge creates through the Chat API.
type OutgoingMessage struct {
	Text    string       `json:"text,omitempty"`
	CardsV2 []CardWithID `json:"cardsV2,omitempty"`
}
```

```go
	var s chat_app_dto.State
	var space, user, sender sql.NullString
	var activated sql.NullTime
	if err := row.Scan(&s.Email, &space, &user, &activated, &sender); err != nil {
		return chat_app_dto.State{}, err
	}
	s.SpaceID = space.String
	s.ChatUserID = user.String
	if activated.Valid {
		t := activated.Time
		s.ActivatedDate = &t
	}
	if sender.Valid {
		v := sender.String
		s.Sender = &v
	}
	return s, nil

func (r *repositoryImpl) FindByEmail(ctx context.Context, email string) (chat_app_dto.State, bool, error) {

	s, err := scanState(row)
	if errors.Is(err, sql.ErrNoRows) {
		return chat_app_dto.State{Email: email}, false, nil
	}
	if err != nil {
		return chat_app_dto.State{}, false, err
	}
	return s, true, nil
}

		return nil, err

	out := map[string]chat_app_dto.State{}
	for rows.Next() {
		s, err := scanState(rows)
		if err != nil {
			return nil, err
		}
		out[strings.ToLower(s.Email)] = s
	}
	return out, rows.Err()
}
```

```go
	if err := json.Unmarshal(b, &many); err != nil {
		return err
	}
	*a = many
	return nil
}

type jwtClaims struct {
	Iss string   `json:"iss"`
	Aud audience `json:"aud"`
	Exp int64    `json:"exp"`
}

func invalid(format string, args ...any) error {
	return fmt.Errorf("%w: %s", ErrInvalidToken, fmt.Sprintf(format, args...))
}

func decodeSegment(seg string, v any) error {
	b, err := base64.RawURLEncoding.DecodeString(seg)
	if err != nil {
		return err
	}
	return json.Unmarshal(b, v)
}

// Verify checks signature, issuer, audience and expiry of token.
func (v *Verifier) Verify(ctx context.Context, token string) error {
	parts := strings.Split(token, ".")
	if len(parts) != 3 {
		return invalid("malformed token")
	}
	var h jwtHeader
	if err := decodeSegment(parts[0], &h); err != nil {
		return invalid("header: %v", err)
	}
	if h.Alg != "RS256" {
		return invalid("unexpected alg %q", h.Alg)
	}
	var c jwtClaims
	if err := decodeSegment(parts[1], &c); err != nil {
		return invalid("claims: %v", err)
	}
	sig, err := base64.RawURLEncoding.DecodeString(parts[2])
	if err != nil {
		return invalid("signature encoding: %v", err)
	}
	key, err := v.key(ctx, h.Kid)
	if err != nil {
		return err
	}
	sum := sha256.Sum256([]byte(parts[0] + "." + parts[1]))
	if err := rsa.VerifyPKCS1v15(key, crypto.SHA256, sum[:], sig); err != nil {
		return invalid("bad signature")
	}
	if c.Iss != ChatIssuer {
		return invalid("issuer %q", c.Iss)
	}
	audOK := false
	for _, a := range c.Aud {
		if a == v.audience {
			audOK = true
		}
	}
	if !audOK {
		return invalid("audience %v", []string(c.Aud))
	}
	if v.now().After(time.Unix(c.Exp, 0).Add(clockSkew)) {
		return invalid("expired")
	}
	return nil
}

func (v *Verifier) key(ctx context.Context, kid string) (*rsa.PublicKey, error) {
	v.mu.Lock()
	defer v.mu.Unlock()
	if k, ok := v.keys[kid]; ok && v.now().Before(v.expires) {
		return k, nil
	}
	if err := v.refresh(ctx); err != nil {
		return nil, err
	}
	if k, ok := v.keys[kid]; ok {
		return k, nil
	}
	return nil, invalid("unknown kid %q", kid)
}

// refresh reloads the certificates; the caller holds v.mu.
func (v *Verifier) refresh(ctx context.Context) error {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, v.certsURL, nil)
	if err != nil {
		return fmt.Errorf("chat_app_auth: build certs request: %w", err)
	}
	resp, err := v.client.Do(req)
	if err != nil {
		return fmt.Errorf("chat_app_auth: fetch certs: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return fmt.Errorf("chat_app_auth: fetch certs: status %d", resp.StatusCode)
	}
	var raw map[string]string
	if err := json.NewDecoder(resp.Body).Decode(&raw); err != nil {
		return fmt.Errorf("chat_app_auth: decode certs: %w", err)
	}
	keys := make(map[string]*rsa.PublicKey, len(raw))
	for kid, certPEM := range raw {
		block, _ := pem.Decode([]byte(certPEM))
		if block == nil {
			continue
		}
		cert, err := x509.ParseCertificate(block.Bytes)
		if err != nil {
			continue
		}
		if pub, ok := cert.PublicKey.(*rsa.PublicKey); ok {
			keys[kid] = pub
		}
	}
	ttl := defaultCertTTL
	if m := maxAgePattern.FindStringSubmatch(resp.Header.Get("Cache-Control")); m != nil {
		if secs, err := strconv.Atoi(m[1]); err == nil && secs > 0 {
			ttl = time.Duration(secs) * time.Second
		}
	}
	v.keys = keys
	v.expires = v.now().Add(ttl)
	return nil
}
```

```go
// the recipient chose (Tech Spec "Nudge sebagai Google Chat App", AC1.x):
// the Nudge Chat App or the operator account's DM. Any app failure falls
// back once to the operator DM — except for the operator account itself,
// which can't DM itself.

	"context"
	"errors"
	"fmt"
	"log"

// ErrRecipientNotActivated means no sender can reach the recipient (e.g.
// the operator account before activating Nudge). Callers record it as
// skipped, not failed.
var ErrRecipientNotActivated = errors.New("chat_delivery: recipient has no usable chat sender")

const inviteFooter = "\n\n💡 Pesan ini bisa datang langsung dari app Nudge — aktifkan di https://nudge.labcup.tech"

type StateStore interface {
	FindByEmail(ctx context.Context, email string) (chat_app_dto.State, bool, error)
	ClearByEmail(ctx context.Context, email string) error
}

type AppSender interface {
	CreateMessage(ctx context.Context, space string, msg chat_app_dto.OutgoingMessage) error
}

type LegacySender interface {
	SendPlainTextDM(ctx context.Context, email, text string) error
}

// CardBuilder turns a reminder into the app message (fallback text + cards).
type CardBuilder func(msg google_chat_notifier_dto.ReminderMessage) (string, []chat_app_dto.CardWithID)

type Deps struct {
	Store StateStore
	// App is nil when the Chat App is not configured.
	App AppSender
	// Legacy is nil when the operator OAuth is not configured.
	Legacy        LegacySender
	OperatorEmail string
	OperatorName  string
	// Cards is nil until phase 3: the app then gets the plain text.
	Cards CardBuilder
}

type Router struct {
	d Deps
}

func New(d Deps) *Router {
	if d.OperatorName == "" {
		d.OperatorName = "operator Nudge"
	}
	return &Router{d: d}
}

// IsOperator reports whether email is the operator (OAuth / bot) account.
func IsOperator(email, operatorEmail string) bool {
	return operatorEmail != "" && strings.EqualFold(strings.TrimSpace(email), strings.TrimSpace(operatorEmail))
}

// EffectiveSender is the sender a message to this user goes through.
func EffectiveSender(s chat_app_dto.State, isOperator, appEnabled bool) string {
	switch {
	case !appEnabled:
		return chat_app_dto.SenderOperator
	case isOperator:
		return chat_app_dto.SenderNudge
	case s.Sender != nil && *s.Sender == chat_app_dto.SenderOperator:
		return chat_app_dto.SenderOperator
	case s.Sender != nil:
		return chat_app_dto.SenderNudge
	case s.Activated():
		return chat_app_dto.SenderNudge
	}
	return chat_app_dto.SenderOperator
}

func (r *Router) SendReminder(ctx context.Context, msg google_chat_notifier_dto.ReminderMessage) error {
	text := google_chat_notifier_service.BuildMessageText(msg)
	appMsg := chat_app_dto.OutgoingMessage{Text: text}
	if r.d.Cards != nil {
		appMsg.Text, appMsg.CardsV2 = r.d.Cards(msg)
	}
	return r.deliver(ctx, msg.AssigneeEmail, text, appMsg)
}

func (r *Router) SendPlainTextDM(ctx context.Context, email, text string) error {
	return r.deliver(ctx, email, text, chat_app_dto.OutgoingMessage{Text: text})
}

func (r *Router) deliver(ctx context.Context, email, text string, appMsg chat_app_dto.OutgoingMessage) error {
	isOp := IsOperator(email, r.d.OperatorEmail)
	state, _, err := r.d.Store.FindByEmail(ctx, email)
	if err != nil {
		return fmt.Errorf("chat_delivery: load chat state of %s: %w", email, err)
	}
	appEnabled := r.d.App != nil

	if EffectiveSender(state, isOp, appEnabled) == chat_app_dto.SenderOperator {
		if isOp || r.d.Legacy == nil {
			return ErrRecipientNotActivated
		}
		if appEnabled && state.Sender == nil && !state.Activated() {
			text += inviteFooter
		}
		return r.d.Legacy.SendPlainTextDM(ctx, email, text)
	}

	appErr := ErrRecipientNotActivated
	if state.Activated() {
		err := r.d.App.CreateMessage(ctx, state.SpaceID, appMsg)
		if err == nil {
			return nil
		}
		appErr = err
		if chat_app_client.IsGone(err) {
			if clearErr := r.d.Store.ClearByEmail(ctx, email); clearErr != nil {
				log.Printf("chat_delivery: forgetting the Chat App DM of %s failed: %v", email, clearErr)
			}
		}
	}
	if isOp || r.d.Legacy == nil {
		return appErr
	}

	log.Printf("chat_delivery: app send to %s failed, falling back to the operator DM: %v", email, appErr)
	footer := fmt.Sprintf("\n\n_(dikirim lewat akun %s karena app Nudge sedang gangguan)_", r.d.OperatorName)
	if err := r.d.Legacy.SendPlainTextDM(ctx, email, text+footer); err != nil {
		if errors.Is(appErr, ErrRecipientNotActivated) {
			return err
		}
		return errors.Join(appErr, err)
	}
	return nil
}
```

```go
	defaultCommandTimeout = 25 * time.Second
	asyncCommandTimeout   = 2 * time.Minute

	processingText    = "⏳ Sedang diproses…"
	errorText         = "Terjadi kesalahan, coba lagi."
	notRegisteredText = "Kamu belum terdaftar di Nudge. Login dulu ke https://nudge.labcup.tech lalu kirim pesan apa saja di sini."

type UserFinder interface {
	FindUserByEmail(ctx context.Context, email string) (*registration_dto.User, bool, error)
}

type StateStore interface {
	FindByEmail(ctx context.Context, email string) (chat_app_dto.State, bool, error)
	SetSpace(ctx context.Context, email, spaceID, chatUserID string) error
	ClearBySpace(ctx context.Context, spaceID string) error
}

type CommandRouter interface {
	RouteCommand(ctx context.Context, senderEmail, messageText string) string
}

type AppSender interface {
	CreateMessage(ctx context.Context, space string, msg chat_app_dto.OutgoingMessage) error

	States   StateStore
	Commands CommandRouter
	App      AppSender
	// CommandTimeout bounds the synchronous reply (Google waits ~30s); 0 = 25s.
	CommandTimeout time.Duration

type Service interface {
	Handle(ctx context.Context, ev chat_app_dto.Event) chat_app_dto.Response
}

type service struct {
	d Deps
}

func New(d Deps) Service {
	if d.CommandTimeout <= 0 {
		d.CommandTimeout = defaultCommandTimeout
	}
	return &service{d: d}
}

func (s *service) Handle(ctx context.Context, ev chat_app_dto.Event) chat_app_dto.Response {
	if !ev.Space.IsDM() {
		return chat_app_dto.Response{}
	}
	switch ev.Type {
	case chat_app_dto.EventAddedToSpace:
		return s.added(ctx, ev)
	case chat_app_dto.EventRemovedFromSpace:
		if err := s.d.States.ClearBySpace(ctx, ev.Space.Name); err != nil {
			log.Printf("chat_app_service: clearing %s failed: %v", ev.Space.Name, err)
		}
		return chat_app_dto.Response{}
	case chat_app_dto.EventMessage:
		return s.message(ctx, ev)
	}
	return chat_app_dto.Response{}
}

// checkUser returns the reply for a sender that may not use Nudge.
func (s *service) checkUser(ctx context.Context, email string) (string, bool) {
	u, found, err := s.d.Users.FindUserByEmail(ctx, email)
	switch {
	case err != nil:
		log.Printf("chat_app_service: FindUserByEmail(%s) failed: %v", email, err)
		return errorText, false
	case !found:
		return notRegisteredText, false
	case !u.FlagActive:
		return inactiveText, false
	}
	return "", true
}

func (s *service) added(ctx context.Context, ev chat_app_dto.Event) chat_app_dto.Response {
	if reply, ok := s.checkUser(ctx, ev.User.Email); !ok {
		return chat_app_dto.Response{Text: reply}
	}
	if err := s.d.States.SetSpace(ctx, ev.User.Email, ev.Space.Name, ev.User.Name); err != nil {
		log.Printf("chat_app_service: storing the DM of %s failed: %v", ev.User.Email, err)
		return chat_app_dto.Response{Text: errorText}
	}
	return chat_app_dto.Response{Text: welcomeText}
}

func (s *service) message(ctx context.Context, ev chat_app_dto.Event) chat_app_dto.Response {
	email := ev.User.Email
	if reply, ok := s.checkUser(ctx, email); !ok {
		return chat_app_dto.Response{Text: reply}
	}
	// The install event may have been missed: remember the DM on any message.
	if state, _, err := s.d.States.FindByEmail(ctx, email); err == nil && state.SpaceID != ev.Space.Name {
		if err := s.d.States.SetSpace(ctx, email, ev.Space.Name, ev.User.Name); err != nil {
			log.Printf("chat_app_service: storing the DM of %s failed: %v", email, err)
		}
	}
	text, usage := commandText(ev.Message)
	if usage != "" {
		return chat_app_dto.Response{Text: usage}
	}
	return chat_app_dto.Response{Text: s.runCommand(email, ev.Space.Name, text)}
}

// commandText is the /n-* text a message stands for; a non-empty second
// value is a usage hint to reply with instead (slash commands, Task 13).
func commandText(msg *chat_app_dto.Message) (string, string) {
	if msg == nil {
		return "", ""
	}
	if arg := strings.TrimSpace(msg.ArgumentText); arg != "" {
		return arg, ""
	}
	return strings.TrimSpace(msg.Text), ""
}

// runCommand answers within CommandTimeout; slower commands reply
// "processing" now and post the result through the app later.
func (s *service) runCommand(email, space, text string) string {
	done := make(chan string, 1)
	go func() {
		ctx, cancel := context.WithTimeout(context.Background(), asyncCommandTimeout)
		defer cancel()
		done <- s.d.Commands.RouteCommand(ctx, email, text)
	}()
	select {
	case out := <-done:
		return out
	case <-time.After(s.d.CommandTimeout):
		go func() {
			out := <-done
			if s.d.App == nil {
				return
			}
			ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
			defer cancel()
			if err := s.d.App.CreateMessage(ctx, space, chat_app_dto.OutgoingMessage{Text: out}); err != nil {
				log.Printf("chat_app_service: late reply to %s failed: %v", email, err)
			}
		}()
		return processingText
	}
}
```

```go
// Package chat_app_handler serves POST /api/chat/events — the Nudge Chat
// App's HTTP endpoint (Tech Spec "Nudge sebagai Google Chat App", AC2.1, AC2.7).
package chat_app_handler

import (
	"context"
	"log"
	"net/http"
	"strings"

	"github.com/gin-gonic/gin"

	"nudge/modules/chat_app/chat_app_dto"
	"nudge/modules/chat_app/chat_app_service"
)

type TokenVerifier interface {
	Verify(ctx context.Context, token string) error
}

type Handler struct {
	verifier TokenVerifier
	svc      chat_app_service.Service
}

func New(verifier TokenVerifier, svc chat_app_service.Service) *Handler {
	return &Handler{verifier: verifier, svc: svc}
}

func (h *Handler) Events(c *gin.Context) {
	token, ok := strings.CutPrefix(c.GetHeader("Authorization"), "Bearer ")
	if !ok || token == "" {
		c.JSON(http.StatusUnauthorized, gin.H{"error": "unauthorized"})
		return
	}
	if err := h.verifier.Verify(c.Request.Context(), token); err != nil {
		log.Printf("chat_app_handler: rejected event: %v", err)
		c.JSON(http.StatusUnauthorized, gin.H{"error": "unauthorized"})
		return
	}
	var ev chat_app_dto.Event
	if err := c.ShouldBindJSON(&ev); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"error": "invalid event"})
		return
	}
	defer func() {
		if r := recover(); r != nil {
			log.Printf("chat_app_handler: %s event panicked: %v", ev.Type, r)
			c.JSON(http.StatusOK, chat_app_dto.Response{Text: "Terjadi kesalahan, coba lagi."})
		}
	}()
	c.JSON(http.StatusOK, h.svc.Handle(c.Request.Context(), ev))
}
```

```go
import (
	"context"
	"encoding/base64"
	"encoding/json"
	"errors"

	"strings"

	"golang.org/x/oauth2"
	"golang.org/x/oauth2/google"

	"nudge/modules/chat_app/chat_app_client"
)

// OAuth is the consent round-trip of the one-click activation.
type OAuth interface {
	AuthCodeURL(state, loginHint string) string
	// Exchange returns the user's access token and the email it belongs to.
	Exchange(ctx context.Context, code string) (accessToken, email string, err error)
}

type googleOAuth struct {
	cfg *oauth2.Config
}

// NewGoogleOAuth reuses the Sign-In OAuth client; redirectURL must be
// registered on it.
func NewGoogleOAuth(clientID, clientSecret, redirectURL string) OAuth {
	return &googleOAuth{cfg: &oauth2.Config{
		ClientID:     clientID,
		ClientSecret: clientSecret,
		RedirectURL:  redirectURL,
		Endpoint:     google.Endpoint,
		Scopes:       []string{"openid", "email", chat_app_client.SpacesCreateScope},
	}}
}

func (g *googleOAuth) AuthCodeURL(state, loginHint string) string {
	return g.cfg.AuthCodeURL(state,
		oauth2.AccessTypeOnline,
		oauth2.SetAuthURLParam("login_hint", loginHint),
		oauth2.SetAuthURLParam("prompt", "consent"),
	)
}

func (g *googleOAuth) Exchange(ctx context.Context, code string) (string, string, error) {
	tok, err := g.cfg.Exchange(ctx, code)
	if err != nil {
		return "", "", fmt.Errorf("chat_app_activation: exchange: %w", err)
	}
	idToken, _ := tok.Extra("id_token").(string)
	email, err := emailFromIDToken(idToken)
	if err != nil {
		return "", "", err
	}
	return tok.AccessToken, email, nil
}

// emailFromIDToken reads the email claim. The ID token came straight from
// Google's token endpoint over TLS, so its signature needs no re-check.
func emailFromIDToken(idToken string) (string, error) {
	parts := strings.Split(idToken, ".")
	if len(parts) != 3 {
		return "", errors.New("chat_app_activation: no id_token in the token response")
	}
	payload, err := base64.RawURLEncoding.DecodeString(parts[1])
	if err != nil {
		return "", fmt.Errorf("chat_app_activation: id_token payload: %w", err)
	}
	var claims struct {
		Email string `json:"email"`
	}
	if err := json.Unmarshal(payload, &claims); err != nil || claims.Email == "" {
		return "", errors.New("chat_app_activation: id_token has no email")
	}
	return claims.Email, nil
}
```

```go
		}
		sender := in.Sender
		if err := h.d.States.SetSender(ctx, target.Email, &sender); err != nil {
			c.JSON(http.StatusInternalServerError, gin.H{"error": "gagal menyimpan pengirim pesan"})
			return
		}
		if onChange != nil {
			onChange(c, target, user_dashboard_handler.SettingsChange{Kind: user_dashboard_handler.ChangeChatSender})
		}
		h.confirm(target.Email)
		res, err := h.status(ctx, target.Email)
		if err != nil {
			c.JSON(http.StatusInternalServerError, gin.H{"error": "gagal memuat status Nudge App"})
			return
		}
		c.JSON(http.StatusOK, res)
	}
}

// confirm DMs the user through the sender they now use (AC7.5) — detached
// and best-effort.
func (h *Handler) confirm(email string) {
	if h.d.Notify == nil {
		return
	}
	go func() {
		ctx, cancel := context.WithTimeout(context.Background(), notifyTimeout)
		defer cancel()
		if err := h.d.Notify.SendPlainTextDM(ctx, email, confirmText); err != nil {
			log.Printf("chat_app_activation: confirmation DM to %s failed: %v", email, err)
		}
	}()
}

func currentUser(c *gin.Context) auth_dto.Session {
	return c.MustGet(auth_middleware.CurrentUserKey).(auth_dto.Session)
}

func (h *Handler) cookiePath() string {
	if h.d.CookiePath == "" {
		return "/"
	}
	return h.d.CookiePath
}

// Activate starts the one-click activation (AC3.2). Mounted behind
// AuthMiddlewareAPI.
func (h *Handler) Activate(c *gin.Context) {
	then := ""
	if c.Query("then") == "sender" {
		then = "sender"
	}
	if h.d.OAuth == nil || !h.d.AppEnabled {
		h.finish(c, then, "failed")
		return
	}
	b := make([]byte, 32)
	if _, err := rand.Read(b); err != nil {
		h.finish(c, then, "failed")
		return
	}
	state := base64.RawURLEncoding.EncodeToString(b)
	c.SetSameSite(http.SameSiteLaxMode)
	c.SetCookie(stateCookieName, state+"|"+then, stateCookieMaxAge, h.cookiePath(), "", true, true)
	c.Redirect(http.StatusFound, h.d.OAuth.AuthCodeURL(state, currentUser(c).Email))
}

// Callback finishes it (AC3.3, AC3.4). The user's token is used once and
// dropped.
func (h *Handler) Callback(c *gin.Context) {
	raw, _ := c.Cookie(stateCookieName)
	c.SetSameSite(http.SameSiteLaxMode)
	c.SetCookie(stateCookieName, "", -1, h.cookiePath(), "", true, true)
	state, then, _ := strings.Cut(raw, "|")

	if raw == "" || state == "" || state != c.Query("state") {
		h.finish(c, then, "failed")
		return
	}
	if e := c.Query("error"); e != "" {
		if e == "access_denied" {
			h.finish(c, then, "cancelled")
		} else {
			h.finish(c, then, "blocked")
		}
		return
	}
	if h.d.OAuth == nil {
		h.finish(c, then, "failed")
		return
	}
	ctx := c.Request.Context()
	session := currentUser(c)
	token, email, err := h.d.OAuth.Exchange(ctx, c.Query("code"))
	if err != nil {
		log.Printf("chat_app_activation: exchange for %s failed: %v", session.Email, err)
		h.finish(c, then, "failed")
		return
	}
	if !strings.EqualFold(email, session.Email) {
		h.finish(c, then, "mismatch")
		return
	}
	space, err := h.d.SetupDM(ctx, token)
	if err != nil {
		log.Printf("chat_app_activation: spaces:setup for %s failed: %v", session.Email, err)
		if chat_app_client.IsGone(err) {
			h.finish(c, then, "not_allowed")
		} else {
			h.finish(c, then, "failed")
		}
		return
	}
	if err := h.d.States.SetSpace(ctx, session.Email, space, ""); err != nil {
		h.finish(c, then, "failed")
		return
	}
	if then == "sender" {
		nudge := chat_app_dto.SenderNudge
		if err := h.d.States.SetSender(ctx, session.Email, &nudge); err != nil {
			log.Printf("chat_app_activation: saving sender of %s failed: %v", session.Email, err)
		}
	}
	h.finish(c, then, "activated")
}

func (h *Handler) finish(c *gin.Context, then, code string) {
	path := "/dashboard"
	if then == "sender" {
		path = "/dashboard/settings"
	}
	c.Redirect(http.StatusFound, h.d.AppURL+path+"?chatApp="+code)
}
```

```ts
import { Injectable, inject, signal } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable, tap } from 'rxjs';
import { environment } from '../../../environments/environment';
import { ToastTone } from '../ui/toast.service';

export type ChatSender = 'nudge' | 'operator';

/** GET /api/dashboard/chat-app/status (Tech Spec "Nudge sebagai Google Chat App"). */
export interface ChatAppStatus {
  enabled: boolean;
  activated: boolean;
  activatedDate: string | null;
  oneClick: boolean;
  sender: ChatSender | null;
  effectiveSender: ChatSender;
  operatorAccount: boolean;
  operatorName: string;
}

/** Toasts for the ?chatApp=<code> the activation callback redirects with. */
export const CHAT_APP_RESULTS: Record<string, { tone: ToastTone; title: string; message: string }> = {
  activated: { tone: 'success', title: 'Nudge App aktif', message: 'Notifikasi bisa datang langsung dari app Nudge.' },
  cancelled: { tone: 'info', title: 'Aktivasi dibatalkan', message: 'Kamu bisa mengaktifkan lewat cara manual kapan saja.' },
  blocked: { tone: 'error', title: 'Diblokir kebijakan ESB', message: 'Gunakan cara manual: cari "Nudge" di Google Chat lalu Install.' },
  not_allowed: { tone: 'error', title: 'Akunmu belum diizinkan', message: 'Hubungi admin Nudge supaya emailmu ditambahkan ke app Nudge.' },
  mismatch: { tone: 'error', title: 'Akun Google berbeda', message: 'Pilih akun Google yang sama dengan akun login dashboard.' },
  failed: { tone: 'error', title: 'Aktivasi gagal', message: 'Coba lagi, atau gunakan cara manual.' },
};

/** The caller's (or, for admins, another user's) Nudge App status and sender choice. */
@Injectable({ providedIn: 'root' })
export class ChatAppService {
  private http = inject(HttpClient);

  /** The logged-in user's status; null until loaded. */
  readonly status = signal<ChatAppStatus | null>(null);

  private base(userId: number | null): string {
    return userId === null
      ? `${environment.apiBasePath}/api/dashboard`
      : `${environment.apiBasePath}/api/dashboard/admin/users/${userId}`;
  }

  load(userId: number | null = null): Observable<ChatAppStatus> {
    return this.http
      .get<ChatAppStatus>(`${this.base(userId)}/chat-app/status`)
      .pipe(tap((s) => userId === null && this.status.set(s)));
  }

  refresh(): void {
    this.load().subscribe({ error: () => undefined });
  }

  setSender(sender: ChatSender, userId: number | null = null): Observable<ChatAppStatus> {
    return this.http
      .put<ChatAppStatus>(`${this.base(userId)}/settings/chat-sender`, { sender })
      .pipe(tap((s) => userId === null && this.status.set(s)));
  }

  activateUrl(then?: 'sender'): string {
    return `${environment.apiBasePath}/auth/chat-app/activate${then ? `?then=${then}` : ''}`;
  }
}
```

```ts
 * US-7): Nudge app or the operator account, one at a time. userId null =
 * the logged-in user; a number = Admin › User detail.

@Component({
  selector: 'app-chat-sender-card',
  standalone: true,
  imports: [IconComponent, ChatAppManualDialogComponent],
  template: `
    @if (status(); as s) {
      @if (s.enabled) {

            <label class="sender-choice" [class.selected]="s.effectiveSender === 'nudge'" [class.disabled]="nudgeDisabled(s)">
              <input type="radio" name="chat-sender" value="nudge" [checked]="s.effectiveSender === 'nudge'" [disabled]="nudgeDisabled(s)" (click)="pick('nudge', $event)" />
              <span>
                <strong>Nudge (app)</strong>
                <small>Pesan datang dari app Nudge di Google Chat. Mendukung command &amp; tombol aksi.</small>
                @if (s.operatorAccount) {
                  <small>Akun operator tidak bisa menerima DM dari dirinya sendiri.</small>
                }
                @if (!s.activated) {
                  <small>Belum diaktifkan{{ userId === null ? '' : ' oleh user ini' }}.</small>
                }
              </span>
            </label>
            @if (!s.operatorAccount) {
              <label class="sender-choice" [class.selected]="s.effectiveSender === 'operator'">
                <input type="radio" name="chat-sender" value="operator" [checked]="s.effectiveSender === 'operator'" [disabled]="saving()" (click)="pick('operator', $event)" />
                <span>
                  <strong>{{ s.operatorName }}</strong>
                  <small>Pesan datang dari akun {{ s.operatorName }} seperti sebelumnya.</small>
                </span>
              </label>
            }
          </div>
          @if (userId === null && !s.activated && (activateFirst() || s.operatorAccount)) {
            <div class="sender-activate" data-activate-first>
              <p>Aktifkan app Nudge dulu supaya pesan bisa dikirim dari sana.</p>
              @if (s.oneClick) {
                <a class="btn btn-primary btn-sm" [href]="chatApp.activateUrl('sender')">Aktifkan Nudge</a>
              }
              <button type="button" class="btn btn-ghost btn-sm" (click)="manualOpen.set(true)">Cara manual</button>
            </div>
          }
        </section>
        <app-chat-app-manual-dialog [open]="manualOpen()" (closed)="manualOpen.set(false)" />
      }
    }
  `,
})
export class ChatSenderCardComponent implements OnInit {
  @Input() userId: number | null = null;

  protected chatApp = inject(ChatAppService);
  private toast = inject(ToastService);

  protected status = signal<ChatAppStatus | null>(null);
  protected saving = signal(false);
  protected activateFirst = signal(false);
  protected manualOpen = signal(false);

  ngOnInit(): void {
    this.reload();
  }

  /** Coming back from Google Chat (manual install) refreshes the status. */
  @HostListener('window:focus')
  protected onFocus(): void {
    this.reload();
  }

  protected reload(): void {
    this.chatApp.load(this.userId).subscribe({ next: (s) => this.status.set(s), error: () => undefined });
  }

  protected nudgeDisabled(s: ChatAppStatus): boolean {
    return this.saving() || s.operatorAccount || (this.userId !== null && !s.activated);
  }

  protected pick(sender: ChatSender, event: Event): void {
    const s = this.status();
    if (!s || this.saving()) return;
    if (sender === s.effectiveSender) return;
    if (sender === 'nudge' && !s.activated) {
      event.preventDefault();
      this.activateFirst.set(true);
      return;
    }
    event.preventDefault();
    this.saving.set(true);
    this.chatApp.setSender(sender, this.userId).subscribe({
      next: (next) => {
        this.status.set(next);
        this.saving.set(false);
        this.toast.success('Pengirim pesan tersimpan', sender === 'nudge' ? 'Notifikasi dikirim dari app Nudge.' : `Notifikasi dikirim dari akun ${s.operatorName}.`);
      },
      error: (err) => {
        this.saving.set(false);
        this.toast.error('Gagal menyimpan pengirim pesan', err?.error?.error ?? 'Coba lagi.');
      },
    });
  }
}
```

```go
// value is a usage hint to reply with instead.
func commandText(msg *chat_app_dto.Message) (string, string) {
	if msg == nil {
		return "", ""
	}
	if msg.SlashCommand != nil {
		name, ok := slashCommands[string(msg.SlashCommand.CommandID)]
		if !ok {
			return "/n-help", ""
		}
		args := strings.TrimSpace(msg.ArgumentText)
		if name == "/n-geser" {
			fields := strings.Fields(args)
			if len(fields) != 2 {
				return "", geserUsage
			}
			return fmt.Sprintf("/n-geser<%s>:%s", strings.ToUpper(fields[0]), strings.ToLower(fields[1])), ""
		}
		if args != "" {
			return name + " " + args, ""
		}
		return name, ""
	}
	if arg := strings.TrimSpace(msg.ArgumentText); arg != "" {
		return arg, ""
	}
	return strings.TrimSpace(msg.Text), ""
}
```

```go
package chat_app_card
```

```go
// GeserFunction is the action name of the move buttons (CARD_CLICKED).
const GeserFunction = "geser"

var boldPattern = regexp.MustCompile(`\*([^*\n]+)\*`)

// ToCardHTML turns Chat markup (*bold*, \n) into the HTML subset card
// text paragraphs render, escaping everything else.
func ToCardHTML(chatText string) string {
	s := html.EscapeString(chatText)
	s = boldPattern.ReplaceAllString(s, "<b>$1</b>")
	return strings.ReplaceAll(s, "\n", "<br>")
}

type move struct {
	label, target string
}

// Build is a chat_delivery.CardBuilder: the short text shown in
// notifications plus one card.
func Build(msg google_chat_notifier_dto.ReminderMessage) (string, []chat_app_dto.CardWithID) {
	var short, body string
	var moves []move
	if msg.Text != "" {
		// Status change notice (AC6.2): Jira link only.
		short, body, _ = strings.Cut(msg.Text, "\n")
		if body == "" {
			body = short
		}
	} else {
		short = google_chat_notifier_service.MessageHeader(msg)
		body = google_chat_notifier_service.BuildMessageText(msg)
		// Same targets /n-geser supports; a ticket in review only goes back.
		if !msg.InReview {
			moves = append(moves, move{"Geser ke Code Review", "code-review"})
		}
		moves = append(moves, move{"Geser ke To Do", "todo"})
	}

	var bs []chat_app_dto.Button
	for _, m := range moves {
		bs = append(bs, chat_app_dto.Button{Text: m.label, OnClick: chat_app_dto.OnClick{Action: &chat_app_dto.OnClickAction{
			Function: GeserFunction,
			Parameters: []chat_app_dto.ActionParameter{
				{Key: "issueKey", Value: msg.IssueKey},
				{Key: "target", Value: m.target},
			},
		}}})
	}
	if msg.IssueURL != "" {
		bs = append(bs, chat_app_dto.Button{Text: "Buka di Jira", OnClick: chat_app_dto.OnClick{OpenLink: &chat_app_dto.OpenLink{URL: msg.IssueURL}}})
	}

	widgets := []chat_app_dto.Widget{{TextParagraph: &chat_app_dto.TextParagraph{Text: ToCardHTML(body)}}}
	if len(bs) > 0 {
		widgets = append(widgets, chat_app_dto.Widget{ButtonList: &chat_app_dto.ButtonList{Buttons: bs}})
	}
	card := chat_app_dto.Card{
		Header:   &chat_app_dto.CardHeader{Title: msg.IssueKey, Subtitle: msg.IssueSummary},
		Sections: []chat_app_dto.Section{{Widgets: widgets}},
	}
	return short, []chat_app_dto.CardWithID{{CardID: "nudge-" + msg.IssueKey, Card: card}}
}

// StripActions returns a copy of card without action buttons (links stay)
// and with status appended as its last section (AC6.3, AC6.4).
func StripActions(card chat_app_dto.Card, status string) chat_app_dto.Card {
	out := chat_app_dto.Card{Header: card.Header}
	for _, s := range card.Sections {
		var ws []chat_app_dto.Widget
		for _, w := range s.Widgets {
			if w.ButtonList == nil {
				ws = append(ws, w)
				continue
			}
			var keep []chat_app_dto.Button
			for _, b := range w.ButtonList.Buttons {
				if b.OnClick.Action == nil {
					keep = append(keep, b)
				}
			}
			if len(keep) > 0 {
				ws = append(ws, chat_app_dto.Widget{ButtonList: &chat_app_dto.ButtonList{Buttons: keep}})
			}
		}
		out.Sections = append(out.Sections, chat_app_dto.Section{Widgets: ws})
	}
	out.Sections = append(out.Sections, chat_app_dto.Section{Widgets: []chat_app_dto.Widget{{TextParagraph: &chat_app_dto.TextParagraph{Text: status}}}})
	return out
}
```
